from pathlib import Path
import os,subprocess,sys,shutil,json,hashlib
root=Path(__file__).parent
wd=root/'worktree'
old=Path('/home/pompote/oldata/victor/morpion-scratch/anemone-chipiron-integration-20261024')
count=int(sys.argv[1]); bench=root/'bench'; logs=root/'logs'
env=dict(os.environ, PYTHONDONTWRITEBYTECODE='1', PYTHONPATH=str(wd), CHIPIRON_TEST_INSTALLED='1', TMPDIR=str(root/'tmp'), XDG_CACHE_HOME=str(root/'cache'), MPLCONFIGDIR=str(root/'cache/matplotlib'), CHIPIRON_OUTPUT_DIR=str(root/'tmp/output'), QT_QPA_PLATFORM='offscreen', ML_FLOW_URI_PATH='sqlite:///'+str(root/'tmp/mlruns.db'), ML_FLOW_URI_PATH_TEST='sqlite:///'+str(root/'tmp/mlruns-test.db'))
def run(py, label, args):
    print(label, flush=True)
    with (logs/(label+'.log')).open('w') as out:
        subprocess.run(['/usr/bin/time','-v','-o',str(logs/(label+'.time')),str(py),'-m','docs.research.bounded_anemone_integration.measure',*map(str,args)],cwd=wd,env=env,stdout=out,stderr=subprocess.STDOUT,check=True)
source=bench/f'source{count}'
run(root/'venv/bin/python',f'prepare{count}',[source,'prepare','--count',count])
for label, parent in [('control',old),('candidate',root)]:
    destination=bench/f'{label}{count}';(destination/'work').mkdir(parents=True)
    shutil.copytree(source/'work/source-checkpoint',destination/'work/source-checkpoint')
    shutil.copyfile(source/'prepare.json',destination/'prepare.json')
    run(parent/'venv/bin/python',f'{label}{count}',[destination,'refresh','--count',count,'--accepted-model-workspace',bench/'model/work'])
control=json.loads((bench/f'control{count}/refresh.json').read_text());candidate=json.loads((bench/f'candidate{count}/refresh.json').read_text())
keys=['checkpoint_sha256','science_sha256','rng_sha256','subsequent_growth','model_sha256','nodes','edges']
parity={k:control[k]==candidate[k] for k in keys}
def files(p):
    return {str(f.relative_to(p)):hashlib.sha256(f.read_bytes()).hexdigest() for f in p.rglob('*') if f.is_file()}
for name in ['refreshed-checkpoint','science']:
    parity[name+'_files']=files(bench/f'control{count}/work'/name)==files(bench/f'candidate{count}/work'/name)
(bench/f'parity{count}.json').write_text(json.dumps(parity,indent=2)+'\n')
print(parity,flush=True)
assert all(parity.values())
