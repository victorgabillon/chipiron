import argparse,json,os,subprocess

case=root/'bench'/('restore-'+a.format)
if a.mode=='prepare':
 from tests.environments.test_morpion_committed_evaluator_restore import committed_neural

 paths,state,current,args=committed_neural(case,a.format)

else:
 from chipiron.environments.morpion.bootstrap.runtime.runner import AnemoneMorpionSearchRunner,AnemoneMorpionSearchRunnerArgs,default_search_args
 from chipiron.environments.morpion.bootstrap.config import MorpionBootstrapRolloutConfig

 r.load_or_create(data['checkpoint'],data['bundle'])

  from chipiron.environments.morpion.bootstrap.reevaluation_transaction import GROWTH_NUMERIC_POLICY
  adapter.numeric_policy=GROWTH_NUMERIC_POLICY
 row=adapter.evaluate_node_window([node])[0]
 data.update(node_id=node.node_id,depth=node.depth,fresh_score_hex=row.direct_value.hex())
 (case/(a.mode+'.json')).write_text(json.dumps(data,indent=2)+'\n')
